### 3.59

实现了两个64位数在128位表示下的乘法

```c
void store_prod(int128_t *dest, long x, long y) {
    long rax = y;
    long rdx = y >> 63;
    long rcx = x;
    rcx = rcx >> 63;
    rcx = rcx * rax;
    rdx = rdx * x;
    rcx = rcx + rdx;

    unsigned __int128 prod = (unsigned __int128)((unsigned long)rax * (unsigned long)x);
    rax = (long)prod;
    rdx = (long)(prod >> 64);

    rdx = rdx + rcx;
    dest->low  = rax;
    dest->high = rdx;
}
```

long switch_prob(long x, long n)
{
    long result = x;
    switch (n) {
    case 60:
    case 62:
        result = x * 8;
        break;
    case 63:
        result = x >> 3;
        break;
    case 64:
        x = (x << 4) - x;
    case 65:
        x = x * x;

}
